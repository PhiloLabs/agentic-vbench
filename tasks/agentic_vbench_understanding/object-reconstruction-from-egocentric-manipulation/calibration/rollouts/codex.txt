+def norm(a):
+    l = math.sqrt(sum(x * x for x in a))

+def cross(a, b):
+    return (
+        a[1] * b[2] - a[2] * b[1],
+        a[2] * b[0] - a[0] * b[2],
+        a[0] * b[1] - a[1] * b[0],
+    )

+def sub(a, b):
+    return (a[0] - b[0], a[1] - b[1], a[2] - b[2])

+
+def add(a, b):
+    return (a[0] + b[0], a[1] + b[1], a[2] + b[2])

+def mul(a, s):
+    return (a[0] * s, a[1] * s, a[2] * s)

+def frame_between(p0, p1):
+    w = norm(sub(p1, p0))
+    ref = (0, 0, 1) if abs(w[2]) < 0.9 else (0, 1, 0)
+    u = norm(cross(w, ref))
+    v = cross(u, w)
+    return u, v, w

+def tube_between(mesh, p0, p1, r0, r1=None, seg=24, cap=True):
+    if r1 is None:
+        r1 = r0
+    u, v, _ = frame_between(p0, p1)
+    a = []
+    b = []
+    for i in range(seg):
+        t = 2 * math.pi * i / seg
+        d = add(mul(u, math.cos(t)), mul(v, math.sin(t)))
+        a.append(mesh.add_v(add(p0, mul(d, r0))))
+        b.append(mesh.add_v(add(p1, mul(d, r1))))

+            mesh.add_tri(ca, a[i], a[j])
+            mesh.add_tri(cb, b[j], b[i])

+def ell_cylinder(mesh, center, rx, ry, zmin, zmax, seg=48, cap=True):
