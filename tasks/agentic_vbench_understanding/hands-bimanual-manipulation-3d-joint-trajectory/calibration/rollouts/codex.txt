    return (c*v[0]-s*v[1], s*v[0]+c*v[1])
def add(a,b): return (a[0]+b[0],a[1]+b[1],a[2]+b[2])
def mul2(v,s,z=0): return (v[0]*s,v[1]*s,z)

def backproject(clip,u,v,z):
    # projectionParams are [f, cx, cy, distortion...]; hands are near image centre enough
    # for this first-order inverse to be stable in metres.
    f,cx,cy=cams[clip]['projectionParams'][:3]
    return ((u-cx)/f*z, (v-cy)/f*z, z)

def gen(clip, frame):
    uw,vw,ut,vt,z,pose=P[clip][frame]
    cfg=pose_cfg[pose]
    wrist=backproject(clip,uw,vw,z)
    d_img=norm((ut-uw, vt-vw))
    # Convert image direction to camera x/y metric axes; +y is down, same as image.
    d=(d_img[0], d_img[1])
    side=(d[1], -d[0])  # for upward fingers, positive side is image/camera right
    palm_len=0.075 if pose in ('keyboard','mouse') else 0.066
    pts={5:wrist}
    # thumb: right hand thumb is on negative side for the common palm-down egocentric view.
    thumb_base=add(wrist, add(mul2(d,palm_len*0.70), mul2(side,-0.046)))
    thdir=norm((d[0]*0.35 + side[0]*-0.94, d[1]*0.35 + side[1]*-0.94))
    th_l=[0.030,0.024,0.023]
    t1=add(thumb_base, mul2(thdir, th_l[0]*cfg['thumb'], cfg['curlz']*0.25))
    t2=add(t1, mul2(thdir, th_l[1]*cfg['thumb'], cfg['curlz']*0.65))
    tt=add(t2, mul2(thdir, th_l[2]*cfg['thumb'], cfg['curlz']))
    pts[6]=t1; pts[7]=t2; pts[0]=tt
    # four fingers
    fmap={'index':(8,9,10,1),'middle':(11,12,13,2),'ring':(14,15,16,3),'pinky':(17,18,19,4)}
    for name,(prox_i,inter_i,dist_i,tip_i) in fmap.items():
        base=add(wrist, add(mul2(d,palm_len), mul2(side,offsets[name])))
        fdir=norm(rot(d, spread_angle[name]*cfg['spread']))
        lens=seg[name]
        # Slightly more bend on ulnar fingers during grasping/cup holding.
        ext=cfg['ext']*(0.92 if name in ('ring','pinky') and pose in ('grasp','cup','cube') else 1.0)
        p=base
        pts[prox_i]=p
        p=add(p, mul2(fdir,lens[0]*ext,cfg['curlz']*0.25)); pts[inter_i]=p
        p=add(p, mul2(fdir,lens[1]*ext,cfg['curlz']*0.60)); pts[dist_i]=p
        p=add(p, mul2(fdir,lens[2]*ext,cfg['curlz'])); pts[tip_i]=p
    return [[round(float(c),4) for c in pts[i]] for i in range(20)]
